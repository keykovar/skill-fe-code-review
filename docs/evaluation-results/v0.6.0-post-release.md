```bash

  https://github.com/keykovar/skill-fe-code-review.git "/new/checkout"
cd "/new/checkout"
pnpm install --frozen-lockfile --store-dir "/new/pnpm-store" --cache-dir "/new/pnpm-cache"
pnpm test
```
